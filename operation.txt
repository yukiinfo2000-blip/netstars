        function aggregateCategoryByQuarter(data, quarter) {
            const months = QUARTER_MAP[quarter] || [4, 5, 6];
            const ali = {}, wechat = {};
            CATEGORIES.forEach(c => { ali[c] = 0; wechat[c] = 0; });
            if (!data) return { aliCategory: ali, wechatCategory: wechat };
            data.forEach(row => {
                if (!row.work_date) return;
                try {
                    const date = new Date(row.work_date);
                    if (isNaN(date.getTime())) return;
                    const m = date.getMonth() + 1;
                    if (!months.includes(m)) return;
                    const ac = row.ali_inquiry_category;
                    if (ac && ali[ac] !== undefined) ali[ac] += 1;
                    const wc = row.wechat_inquiry_category;
                    if (wc && wechat[wc] !== undefined) wechat[wc] += 1;
                } catch (e) { /* ignore */ }
            });
            return { aliCategory: ali, wechatCategory: wechat };
        }

                            display: function(c) { return c.dataset.data[c.dataIndex] > 0; },
                            color: '#ffffff',
                            font: { weight: '700', size: 11 },
                            formatter: function(v, c) {
                                const t = c.dataset.data.reduce((a, b) => a + b, 0);
                                if (t === 0 || v === 0) return '';
                                return (v / t * 100).toFixed(1) + '%';
                            },
                            anchor: 'center',
                            align: 'center'
                        }
                    },
                    cutout: '50%'
                },
                plugins: [ChartDataLabels]
            });
            chartInstances[canvasId + 'Instance'] = chart;
        }

        // ================================================================
        function aggregateKPIByQuarter(data, quarter) {
            const months = QUARTER_MAP[quarter] || [4, 5, 6];
            const result = {};
            months.forEach(m => {
                result[m] = { pms: 0, customer_support: 0, invoice: 0, promo: 0, discover: 0, coupon: 0, wsystem: 0 };
            });
            if (!data) return result;
            data.forEach(row => {
                if (!row.work_date) return;
                try {
                    const date = new Date(row.work_date);
                    if (isNaN(date.getTime())) return;
                    const m = date.getMonth() + 1;
                    if (!months.includes(m)) return;
                    result[m].pms += Number(row.pms_count) || 0;
                    result[m].customer_support += (Number(row.wechat_exchange_count) || 0) + (Number(row.ali_exchange_count) || 0);
                    result[m].invoice += Number(row.invoice_count) || 0;
                    result[m].promo += Number(row.promo_count) || 0;
                    result[m].discover += Number(row.discover_count) || 0;
                    result[m].coupon += Number(row.coupon_count) || 0;
                    result[m].wsystem += Number(row.wsystem_count) || 0;
                } catch (e) { /* ignore */ }
            });
            return result;
        }

        function renderKPIs(quarterlyData, quarter) {
            const grid = document.getElementById('kpiMonthGrid');
            const months = QUARTER_MAP[quarter] || [4, 5, 6];

            Object.keys(chartInstances).forEach(k => {
                if (k.startsWith('kpi_')) destroyChart(k);
            });

            if (!quarterlyData || Object.keys(quarterlyData).length === 0) {
                grid.innerHTML = '<div class="empty-state">📭 データがありません</div>';
                return;
            }

            grid.innerHTML = '';
            months.forEach(month => {
                const md = quarterlyData[month] || { pms: 0, customer_support: 0, invoice: 0, promo: 0, discover: 0, coupon: 0, wsystem: 0 };
                const values = [md.pms || 0, md.customer_support || 0, md.invoice || 0, md.promo || 0, md.discover || 0, md.coupon || 0, md.wsystem || 0];

                const card = document.createElement('div');
                card.className = 'month-card';
                const canvasId = `kpi_${month}`;
                const emojis = ['📋', '💬', '📄', '📦', '🏪', '🎫', '⚙️'];
                card.innerHTML = `<div class="month-title"><span class="month-icon">${emojis[0]}</span>${month}月</div><div class="chart-container"><canvas id="${canvasId}"></canvas></div>`;
                grid.appendChild(card);

                const canvas = document.getElementById(canvasId);
                if (!canvas) return;
                const ctx = canvas.getContext('2d');

                const chart = new Chart(ctx, {
                    type: 'bar',
                    data: {
                        labels: KPI_LABELS,
                        datasets: [{
                            data: values,
                            backgroundColor: COLORS.kpi,
                            borderColor: COLORS.kpiBorder,
                            borderWidth: 1.5,
                            borderRadius: 4,
                            barPercentage: 0.6,
                            categoryPercentage: 0.7
                        }]
                    },
                    options: {
                        indexAxis: 'x',
                        responsive: true,
                        maintainAspectRatio: false,
                        plugins: {
                            legend: { display: false },
                            tooltip: { callbacks: { label: function(c) { return c.raw.toFixed(1); } } },
                            datalabels: {
                                anchor: 'end', align: 'end', color: '#3a4f6a',
                                font: { weight: '700', size: 8 },
                                formatter: function(v) { return v.toFixed(1); },
                                offset: 1
                            }
                        },
                        scales: {
                            y: { min: 0, grid: { color: 'rgba(0,0,0,0.04)' }, ticks: { color: '#6a7f98', stepSize: 1, font: { size: 9 } }, title: { display: false } },
                            x: { grid: { display: false }, ticks: { color: '#3a4f6a', font: { size: 8, weight: '600' }, maxRotation: 25, minRotation: 10 } }
                        },
                        layout: { padding: { top: 14, right: 4, bottom: 4, left: 4 } }
                    },
                    plugins: [ChartDataLabels]
                });
                chartInstances[canvasId] = chart;
            });
        }

        // ================================================================
        let cachedData = null;

        async function loadDashboard() {
            try {
                const data = await fetchData();
                cachedData = data;

                if (!data || data.length === 0) {
                    document.getElementById('tableBody').innerHTML = `<tr><td colspan="13" class="empty-state">📭 データがありません</td></tr>`;
                    document.getElementById('alipayLegend').innerHTML = '<span class="empty-state">データなし</span>';
                    document.getElementById('wechatLegend').innerHTML = '<span class="empty-state">データなし</span>';
                    document.getElementById('kpiMonthGrid').innerHTML = '<div class="empty-state">📭 データがありません</div>';
                    return;
                }

                // 模块1
                const year = parseInt(document.getElementById('yearSelect').value) || 2026;
                const monthlyData = aggregateByMonth(data, year);
                renderBusinessTable(monthlyData);

                // 模块2
                const quarter = document.getElementById('quarterSelect').value || 'Q2';
                const { aliCategory, wechatCategory } = aggregateCategoryByQuarter(data, quarter);
                renderCategoryChart('alipayChart', 'alipayLegend', 'alipayQuarterLabel', aliCategory, quarter);
                renderCategoryChart('wechatChart', 'wechatLegend', 'wechatQuarterLabel', wechatCategory, quarter);

                // 模块3
                const kpiQuarter = document.getElementById('kpiQuarterSelect').value || 'Q2';
                const kpiData = aggregateKPIByQuarter(data, kpiQuarter);
                renderKPIs(kpiData, kpiQuarter);

                // ---- 绑定事件 ----
                document.getElementById('yearSelect').addEventListener('change', function() {
                    const y = parseInt(this.value);
                    const md = aggregateByMonth(cachedData, y);
                    renderBusinessTable(md);
                });

                document.getElementById('quarterSelect').addEventListener('change', function() {
                    const q = this.value;
                    const { aliCategory: ac, wechatCategory: wc } = aggregateCategoryByQuarter(cachedData, q);
                    renderCategoryChart('alipayChart', 'alipayLegend', 'alipayQuarterLabel', ac, q);
                    renderCategoryChart('wechatChart', 'wechatLegend', 'wechatQuarterLabel', wc, q);
                });

                document.getElementById('kpiQuarterSelect').addEventListener('change', function() {
                    const q = this.value;
                    const kd = aggregateKPIByQuarter(cachedData, q);
                    renderKPIs(kd, q);
                });

                console.log('✅ 全てのダッシュボードが正常に読み込まれました');

            } catch (err) {
                console.error('❌ 加载失败:', err);
                document.getElementById('tableBody').innerHTML = `<tr><td colspan="13" class="error-message">⚠️ ${err.message}</td></tr>`;
            }
        }

        document.addEventListener('DOMContentLoaded', function() {
            console.log('📋 DOM loaded, initializing dashboard...');
            loadDashboard();
        });
    </script>
