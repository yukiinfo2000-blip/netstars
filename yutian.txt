        // ============================================================
        // 3. Supabase データ取得

        async function fetchWorkRecords(year) {
            const cacheKey = CACHE_KEY_PREFIX + year;
            const cached = localStorage.getItem(cacheKey);
            if (cached) {
                try {
                    const parsed = JSON.parse(cached);
                    if (parsed._timestamp && Date.now() - parsed._timestamp < 24 * 60 * 60 * 1000) {
                        console.log(`📦 キャッシュから取得 (${year})`);
                        return parsed.data;
                    }
                } catch (e) {}
            }

            console.log(`🌐 Supabase から取得 (${year})`);

            const startDate = `${year}-01-01`;
            const endDate = `${year}-12-31`;

            const supabase = window.supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY);

            const { data, error } = await supabase
                .from(TABLE_NAME)
                .select('date, count, category, success_orders, cancel_orders')
                .gte('date', startDate)
                .lte('date', endDate);

            if (error) {
                console.error('Supabase Error:', error);
                throw new Error(`データ取得エラー: ${error.message}`);
            }

            console.log(`📊 取得データ件数: ${data ? data.length : 0}件`);

            const cacheData = {
                _timestamp: Date.now(),
                data: data || [],
            };
            try {
                localStorage.setItem(cacheKey, JSON.stringify(cacheData));
            } catch (e) {}

            return data || [];
        }

        // ============================================================
        // 4. データ加工
        // ============================================================
        function processRawData(rawData, year) {
            const catMonthly = {};
            const catTotal = {};
            const monthlyTotal = Array(12).fill(0);
            const monthlySuccess = Array(12).fill(0);
            const monthlyCancel = Array(12).fill(0);

            const allCategories = new Set();
            rawData.forEach(record => {
                const category = record.category || 'その他';
                allCategories.add(category);
            });

            if (allCategories.size === 0) {
                allCategories.add('データなし');
            }

            allCategories.forEach(c => {
                catMonthly[c] = Array(12).fill(0);
                catTotal[c] = 0;
            });

            rawData.forEach(record => {
                const date = record.date;
                const count = Number(record.count) || 0;
                const category = record.category || 'その他';
                const success = Number(record.success_orders) || 0;
                const cancel = Number(record.cancel_orders) || 0;

                let monthIdx = -1;
                try {
                    monthIdx = new Date(date).getMonth();
                } catch (e) {
                    console.warn('日付パースエラー:', date);
                    return;
                }

                if (monthIdx >= 0 && monthIdx < 12) {
                    monthlyTotal[monthIdx] += count;
                    monthlySuccess[monthIdx] += success;
                    monthlyCancel[monthIdx] += cancel;
                }

                if (!catMonthly[category]) {
                    catMonthly[category] = Array(12).fill(0);
                    catTotal[category] = 0;
                    allCategories.add(category);
                }

                if (monthIdx >= 0 && monthIdx < 12) {
                    catMonthly[category][monthIdx] += count;
                    catTotal[category] += count;
                }
            });

            const activeCats = Array.from(allCategories).filter(c => catTotal[c] > 0);
            const annualTotal = monthlyTotal.reduce((a, b) => a + b, 0);
            const annualSuccess = monthlySuccess.reduce((a, b) => a + b, 0);
            const annualCancel = monthlyCancel.reduce((a, b) => a + b, 0);

            // ★ 月別問い合わせ率（%）を計算
            // 問い合わせ率 = 問い合わせ件数 ÷ 取引件数 × 100
            // 取引件数 = success + cancel
            const monthlyRate = monthlyTotal.map((count, idx) => {
                const tradeTotal = monthlySuccess[idx] + monthlyCancel[idx];
                if (tradeTotal === 0) return 0;
                return parseFloat(((count / tradeTotal) * 100).toFixed(1));
            });

            const catPercent = {};
            allCategories.forEach(c => {
                catPercent[c] = annualTotal > 0 ? (catTotal[c] / annualTotal) * 100 : 0;
            });

            const sortedCats = [...activeCats].sort((a, b) => catTotal[b] - catTotal[a]);

            console.log(`📊 処理結果: 問い合わせ合計${annualTotal}件, 成功${annualSuccess}件, キャンセル${annualCancel}件`);
            console.log(`📊 月別問い合わせ率:`, monthlyRate);

            return {
                year,
                monthlyTotal,
                monthlySuccess,
                monthlyCancel,
                monthlyRate,
                annualTotal,
                annualSuccess,
                annualCancel,
                catMonthly,
                catTotal,
                catPercent,
                activeCats,
                sortedCats,
                top6: sortedCats.slice(0, 6),
                top3: sortedCats.slice(0, 3),
                rawData,
            };
        }

        // 6. モックデータ生成
        // ============================================================
        function generateMockData(year) {
            const seed = year === 2026 ? 42 : 137;
            const rng = (s) => { let x = Math.sin(s) * 43758.5453; return x - Math.floor(x); };

            const categories = ['商品予約', '受取手順', 'アフター関連', '受取場所', '営業時間', '支払方法'];
            const monthlyTotal = [];
            const monthlySuccess = [];
            const monthlyCancel = [];

            for (let m = 0; m < 12; m++) {
                const base = 22 + Math.floor(rng(seed + m * 7) * 18);
                const success = Math.floor(base * (0.3 + rng(seed + m * 11) * 0.3));
                const cancel = Math.floor(base * (0.05 + rng(seed + m * 13) * 0.1));
                monthlyTotal.push(base);
                monthlySuccess.push(success);
                monthlyCancel.push(cancel);
            }

            const catMonthly = {};
            const catTotal = {};
            categories.forEach(c => { catMonthly[c] = Array(12).fill(0);
                catTotal[c] = 0; });

            for (let m = 0; m < 12; m++) {
                const total = monthlyTotal[m];
                const weights = categories.map((_, i) => 0.5 + rng(seed + m * 13 + i * 31) * 1.2);
                const sumW = weights.reduce((a, b) => a + b, 0);
                let remaining = total;
                const counts = categories.map((_, i) => {
                    if (i === categories.length - 1) return remaining;
                    const val = Math.round((weights[i] / sumW) * total);
                    remaining -= val;
                    return val;
                });
                categories.forEach((cat, i) => {
                    catMonthly[cat][m] = counts[i] || 0;
                    catTotal[cat] += counts[i] || 0;
                });
            }

            const annualTotal = monthlyTotal.reduce((a, b) => a + b, 0);
            const annualSuccess = monthlySuccess.reduce((a, b) => a + b, 0);
            const annualCancel = monthlyCancel.reduce((a, b) => a + b, 0);
            const catPercent = {};
            categories.forEach(c => { catPercent[c] = annualTotal > 0 ? (catTotal[c] / annualTotal) * 100 : 0; });

            const activeCats = categories.filter(c => catTotal[c] > 0);
            const sortedCats = [...activeCats].sort((a, b) => catTotal[b] - catTotal[a]);

            // モック用の月別問い合わせ率も計算
            const monthlyRate = monthlyTotal.map((count, idx) => {
                const tradeTotal = monthlySuccess[idx] + monthlyCancel[idx];
                if (tradeTotal === 0) return 0;
                return parseFloat(((count / tradeTotal) * 100).toFixed(1));
            });

            return {
                year,
                monthlyTotal,
                monthlySuccess,
                monthlyCancel,
                monthlyRate,
                annualTotal,
                annualSuccess,
                annualCancel,
                catMonthly,
                catTotal,
                catPercent,
                activeCats,
                sortedCats,
                top6: sortedCats.slice(0, 6),
                top3: sortedCats.slice(0, 3),
                rawData: [],
            };
        }

        // ============================================================
        // 7. メイン処理
        // ============================================================
        async function loadAndRender(year) {
            loadingOverlay.classList.remove('hidden');
            loadErrorEl.style.display = 'none';
            dataStatusEl.textContent = '⏳ 読込中...';
            dataStatusEl.className = 'data-status';

            try {
                if (SUPABASE_URL === '' || SUPABASE_ANON_KEY === '') {
                    throw new Error('⚠️ SupabaseのURLとANON_KEYを設定してください。');
                }

                const rawData = await fetchWorkRecords(year);
                console.log(`📊 取得データ (${year}):`, rawData.length, '件');

                const processed = processRawData(rawData, year);
                dataCache[year] = processed;
                renderDashboard(year, processed);

            } catch (error) {
                console.error('Error:', error);
                loadErrorEl.textContent = `❌ ${error.message}`;
                loadErrorEl.style.display = 'block';
                dataStatusEl.textContent = '❌ エラー';
                dataStatusEl.className = 'data-status error';

                console.warn('⚠️ モックデータで表示します（デモモード）');
                const mockData = generateMockData(year);
                dataCache[year] = mockData;
                renderDashboard(year, mockData);
                dataStatusEl.textContent = '⚠️ モック表示';
                dataStatusEl.className = 'data-status error';

            } finally {
                loadingOverlay.classList.add('hidden');
            }
        }

        function handleYearChange() {
            const year = parseInt(yearSelect.value, 10);

            } else {

            }
        }

        yearSelect.addEventListener('change', handleYearChange);

        const initialYear = parseInt(urlParams.get('year')) || 2026;
